# Defaults

Generated by `scripts/build_knowledge.py` from `default_config.yaml`; do not edit by hand.

Every value is a starting point, not an acceptance criterion: each laboratory reviews it within its own quality system. An assay file can override a value under `settings:`; a `--config` file overrides the defaults for every run.[^default-config]

```yaml
variants:
  # Exhaustive variant analysis of the intended target: the variant summary and inclusivity are
  # built from EVERY genome assembly of the target taxon in NCBI Datasets (complete and draft),
  # not from BLAST's best hits. Each genome is downloaded, scanned for the reference amplicon and
  # discarded; only the amplicon region plus flanks is kept (cached, so later runs only process
  # assemblies that are new). Needs the reference amplicon (assay 'reference_amplicon', or found
  # via the oligos in the target 'accession').
  source: datasets                 # datasets: every genome assembly (bacteria, most targets)
                                   # blast_partitioned: every NCBI Nucleotide record, BLASTed in
                                   #   accession lists (targets without assemblies, e.g. viruses
                                   #   or single genes; sends the reference amplicon to BLAST)
  current_assemblies_only: true    # the latest version of each assembly
  exclude_atypical: true           # NCBI's "atypical" flag (assembly problems)
  max_assemblies_per_run: 20000    # newest first; the rest follow on later runs (resumable)
  flank_nt: 50                     # bases kept on each side of the amplicon
  seed_length: 16                  # exact k-mers of the reference fragment used to find it
  seed_step: 2                     #   (every 2nd position, both strands; measured 2026-09-29:
                                   #   every 4th missed divergent opa copies, every 2nd did not)
  # blast_partitioned only:
  nucleotide_query: null           # extra Entrez filter for the record list, e.g. "25000:32000[SLEN]"
  blast_max_records_per_run: 2000  # newest first; one BLAST search per blast_records_per_search
  blast_records_per_search: 100    # accessions per search (100 checked live; the maximum here)
  direct_scan_max_length: 200000   # records BLAST did not hit and at most this long are fetched
                                   #   and scanned directly (newest records may not be in BLAST yet)
  direct_scan_batch: 20            # records per EFetch request for that direct scan
  probe_channels: any              # probes with different reporters are separate channels:
                                   #   any = a genome counts as detected if any channel detects it
                                   #   all = only if every channel does
  homopolymer_bulges_detectable: false  # a site differing only by the length of a single-base
                                   #   run (e.g. poly-A 7->8), without any mismatch: false =
                                   #   its class decides (primers: rule R5b, at risk or likely
                                   #   failure); true = detectable. The report shows the count
                                   #   under both rules either way
  multicopy_unassembled: auto      # multi-copy targets: a draft genome (Scaffold/Contig, more
                                   #   than one sequence or a copy cut at a contig end) whose best
                                   #   copy fails, with fewer than half the copies of the complete
                                   #   and chromosome-level genomes of the run (median; at least 5
                                   #   of them, median 2 or more), counts as undetermined "copies
                                   #   possibly unassembled" unless one of those fails the same
                                   #   way (also in the panel command);
  # A located region counts as a copy of the locus when ANY of (a), (b), (c) holds; weaker
  # regions are listed, never judged. Measured (2026-09-28/29): chance regions reach at most 18
  # anchored bases and identity 0.54-0.60; true copies of a divergent species 114+ bases.
  min_anchored_bases: 32           # (a) fragment bases in exact blocks, in order
  min_context_bases: 32            # (b) bases anchored in the sequence on one side of the
                                   #   fragment (context accession), with the fragment touched,
                                   #   flanked on both sides, or cut by a contig end
  min_copy_identity: 0.75          # (c) identity to the reference fragment over the part the
                                   #   copy covers (e.g. the divergent N. gonorrhoeae opa copy:
                                   #   0.80) ...
  min_identity_anchored_bases: 16  #   ... with at least this many fragment bases anchored
  max_indel: 150                   # largest length difference to the reference within one copy
  judge_from_parts: detectable     # genomes whose copies are cut by a contig end (e.g. rRNA
                                   #   operons in draft assemblies) but that carry a detectable
                                   #   site of every role on the cut copies (2+ exact seeds each):
                                   #   detectable = counted as detected (default since 2026-10-01,
                                   #   user: every site was seen whole); undetermined = its own
                                   #   class "detectable from parts", counted as undetermined;
                                   #   off = "cut by a contig end" (undetermined). The sites may
                                   #   come from different copies, so this is never proof of one
                                   #   amplicon
                                   #   off = counted as an escape
```
